cmake_minimum_required(VERSION 3.20)
project(ShotOcr LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Widgets Network)
find_package(PkgConfig REQUIRED)
pkg_check_modules(ATSPI REQUIRED IMPORTED_TARGET atspi-2 gobject-2.0)
add_executable(shotocr linux/main.cpp)
target_link_libraries(shotocr PRIVATE Qt6::Widgets Qt6::Network PkgConfig::ATSPI)
target_compile_options(shotocr PRIVATE -Wall -Wextra)
install(TARGETS shotocr RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
    add_executable(selection_test linux/selection_test.cpp)
    target_link_libraries(selection_test PRIVATE Qt6::Widgets Qt6::Network PkgConfig::ATSPI)
    add_test(NAME selection COMMAND selection_test)
    set_tests_properties(selection PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
add_executable(integration_test EXCLUDE_FROM_ALL linux/integration_test.cpp)
target_link_libraries(integration_test PRIVATE Qt6::Widgets Qt6::Network PkgConfig::ATSPI)
